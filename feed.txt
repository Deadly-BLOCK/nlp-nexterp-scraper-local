function processRawData(data){
  const raw=(data.capturedData||[]).flatMap(c=>c.postLists||[]);
  const byId=Array.from(new Map(raw.map(p=>[sid(p.postId),{
    id:sid(p.postId), teacher:p.userName||'System',
    datetime:p.createdOn,
    content:(p.content||'').replace(/(color|background-color|background)\s*:[^;]+;?/gi,''),
    title:p.title||'', subject:p.subjectName||'General',
    attachments:p.postAttachments||[], imageUrl:p.imageUrl||null
  }])).values())
  .filter(p=>!(p.title+p.content).toLowerCase().includes('resource published in'));
  const seen=new Set();
  return byId.filter(p=>{
    const fp=`${p.teacher}||${p.title}||${stripHtml(p.content)}`;
    if(seen.has(fp)) return false; seen.add(fp); return true;
  }).sort((a,b)=>new Date(b.datetime)-new Date(a.datetime));
}


function filteredPosts(){
  const q=el('search').value.toLowerCase();
  const tf=el('teacherFilter').value;
  const sf=el('subjectFilter').value;
  const range=el('quickRange').value;
  let from=el('dateFrom').value, to=el('dateTo').value;
  if(range!=='all'&&range!=='custom'){const d=new Date();d.setDate(d.getDate()-parseInt(range));from=d.toISOString().split('T')[0];to='';}
  const bl=prefs.blocklist||{teachers:[],subjects:[]};
  return allPosts.filter(p=>{
    const pd=new Date(p.datetime).toISOString().split('T')[0];
if((bl.teachers||[]).includes(p.teacher)) return false;
    if((bl.subjects||[]).includes(p.subject)) return false;
    if(globalBlockWords.length){
      const haystack=`${p.teacher} ${p.subject} ${p.title} ${stripHtml(p.content)}`.toLowerCase();
      if(globalBlockWords.some(w=>haystack.includes(w))) return false;
    }
    if(bookmarksOnly&&!(prefs.bookmarks||[]).includes(sid(p.id))) return false;
    return `${p.teacher} ${p.content} ${p.subject} ${p.title}`.toLowerCase().includes(q)&&
      (!tf||p.teacher===tf)&&(!sf||p.subject===sf)&&(!from||pd>=from)&&(!to||pd<=to);
  });
}

function render(){
  
  el('grid').querySelectorAll('.skeleton').forEach(s=>s.remove());

  const fp=filteredPosts();
  const pins=(prefs.pins||[]).map(sid);
  const pinned=fp.filter(p=>pins.includes(sid(p.id)));
  const unpinned=fp.filter(p=>!pins.includes(sid(p.id)));

  
  if(pinned.length){
    el('pinnedSection').classList.remove('hide');
    reconcileGrid(el('pinnedGrid'), pinned, true);
  } else {
    el('pinnedSection').classList.add('hide');
    el('pinnedGrid').innerHTML='';
  }

  
  displayedCount=Math.min(PAGE_SIZE, unpinned.length);
  const slice=unpinned.slice(0, displayedCount);
  if(!slice.length && !el('grid').querySelector('.card')){
    el('grid').innerHTML=`<div style="text-align:center;padding:48px;color:var(--c-muted);border:1px dashed var(--c-border);border-radius:var(--radius)">No posts found.</div>`;
  } else {
    reconcileGrid(el('grid'), slice, false);
  }

  el('countText').textContent=`Showing ${Math.min(displayedCount,unpinned.length)} of ${fp.length} posts`;
  setupReadObserver();
  updateUnreadBadge();
}

function reconcileGrid(container, posts, forPinned){
  
  container.querySelectorAll('.skeleton').forEach(s=>s.remove());
  
  const ids=new Set(posts.map(p=>sid(p.id)));
  container.querySelectorAll('.card[data-id]').forEach(c=>{
    if(!ids.has(c.dataset.id)) c.remove();
  });
  
  posts.forEach((p,i)=>{
    const id=sid(p.id);
    let card=container.querySelector(`.card[data-id="${id}"]`);
    if(!card){
      
      const wrap=document.createElement('div');
      wrap.innerHTML=buildCard(p, true);
      card=wrap.firstElementChild;
      const ref=container.querySelectorAll('.card[data-id]')[i];
      if(ref) container.insertBefore(card, ref);
      else container.appendChild(card);
    } else {
      
      updateCardState(card, p, forPinned);
      
      const allCards=Array.from(container.querySelectorAll('.card[data-id]'));
      if(allCards.indexOf(card)!==i){
        const ref=allCards[i];
        if(ref&&ref!==card) container.insertBefore(card,ref);
        else if(!ref) container.appendChild(card);
      }
    }
  });
}

function updateCardState(card, p, forPinned){
  const id=sid(p.id);
  const pins=(prefs.pins||[]).map(sid);
  const bms=(prefs.bookmarks||[]).map(sid);
  const reads=(prefs.readIds||[]).map(sid);
  const isUnread=!reads.includes(id);
  const isBookmarked=bms.includes(id);
  const isPinned=pins.includes(id);
  const isToday=todaySubjects.has((p.subject||'').toLowerCase());
  card.classList.toggle('unread',isUnread);

function loadMore(){

  if(displayedCount>=unpinned.length) return;

let readObserver=null;
function setupReadObserver(){
  if(readObserver) readObserver.disconnect();
  readObserver=new IntersectionObserver(entries=>{
    let changed=false;
    entries.forEach(e=>{
      if(e.isIntersecting){
        const id=sid(e.target.dataset.id||'');
        if(id&&!(prefs.readIds||[]).map(sid).includes(id)){
          (prefs.readIds=prefs.readIds||[]).push(id);
          e.target.classList.remove('unread');
          changed=true; readObserver.unobserve(e.target);
        }
      }
    });
    if(changed){updateUnreadBadge();savePrefs();}
  },{threshold:.3});
  document.querySelectorAll('.card[data-id]').forEach(c=>readObserver.observe(c));
}

        e.stopPropagation();
        const pins=(prefs.pins||[]).map(sid);

            render();

        }

      }
      if(action==='bookmark'&&id){
        e.stopPropagation();

        if(idx>-1) prefs.bookmarks=bms.filter((_,i)=>i!==idx);

function renderTodayWidget(td){
  const dayNames=['SUN','MON','TUE','WED','THU','FRI','SAT'];
  const today=dayNames[new Date().getDay()];
  const now=new Date(); const nowMin=now.getHours()*60+now.getMinutes();
  let todayCol=-1;
  for(let c=0;c<td.cols;c++){if(((td.cells[`0,${c}`]||{}).v||'').trim().toUpperCase()===today){todayCol=c;break;}}
  todaySubjects.clear();
  const listEl=el('todayList'); if(!listEl) return;
  if(todayCol===-1){listEl.innerHTML='<div style="color:var(--c-muted);font-size:.76rem">No classes configured</div>';return;}
  const periods=[];
  for(let r=1;r<td.rows;r++){
    const ts=((td.cells[`${r},0`]||{}).v||'').trim();
    const subj=((td.cells[`${r},${todayCol}`]||{}).v||'').trim();
    if(!subj) continue;
    const m=ts.match(/(\d+):?(\d*)/);
    const startMin=m?parseInt(m[1])*60+(parseInt(m[2])||0):-1;
    periods.push({time:ts,subj,startMin}); todaySubjects.add(subj.toLowerCase());
  }
  if(!periods.length){listEl.innerHTML='<div style="color:var(--c-muted);font-size:.76rem">No classes today</div>';return;}
  let ci=-1;
  periods.forEach((p,i)=>{const next=periods[i+1];if(p.startMin<0)return;if(nowMin>=p.startMin&&(!next||nowMin<next.startMin))ci=i;});
  listEl.innerHTML=periods.map((p,i)=>`<div class="today-period${i===ci?' current':''}${p.startMin>-1&&nowMin>p.startMin&&i!==ci?' past':''}">
    <div class="today-dot${i===ci?' live':''}"></div>
    <span class="today-time">${escHtml(p.time)}</span>
    <span class="today-subj">${escHtml(p.subj)}</span>
  </div>`).join('');

let countdownInterval=null;
let nextPeriodData=null; 

function startCountdown(){
  if(countdownInterval) clearInterval(countdownInterval);
  if(!nextPeriodData){el('nextCountdown').classList.add('hide');return;}
  function tick(){
    const now=new Date();
    const nowMin=now.getHours()*60+now.getMinutes();
    const nowSec=now.getSeconds();
    let diffSec=(nextPeriodData.startMin-nowMin)*60-nowSec;
    if(diffSec<0){clearInterval(countdownInterval);el('nextCountdown').classList.add('hide');return;}
    const h=Math.floor(diffSec/3600);
    const m=Math.floor((diffSec%3600)/60);
    const s=diffSec%60;
    let label='';
    if(h>0) label=`${h}h ${m}m`;
    else if(m>0) label=`${m}m ${String(s).padStart(2,'0')}s`;
    else label=`${s}s`;
    el('nextCountdownText').innerHTML=`Next: <strong>${escHtml(nextPeriodData.name)}</strong> in <span class="today-countdown-time">${label}</span>`;
    el('nextCountdown').classList.remove('hide');
  }
  tick();
  countdownInterval=setInterval(tick,1000);
}


const _origRenderTodayWidget=renderTodayWidget;
function renderTodayWidgetWithCountdown(td){
  _origRenderTodayWidget(td);
  
  const now=new Date();
  const nowMin=now.getHours()*60+now.getMinutes();
  const dayNames=['SUN','MON','TUE','WED','THU','FRI','SAT'];
  const today=dayNames[now.getDay()];
  let todayCol=-1;
  for(let c=0;c<td.cols;c++){
    if(((td.cells[`0,${c}`]||{}).v||'').trim().toUpperCase()===today){todayCol=c;break;}
  }
  nextPeriodData=null;
  if(todayCol>-1){
    for(let r=1;r<td.rows;r++){
      const ts=((td.cells[`${r},0`]||{}).v||'').trim();
      const subj=((td.cells[`${r},${todayCol}`]||{}).v||'').trim();
      if(!subj) continue;
      const m=ts.match(/(\d+):?(\d*)/);
      if(!m) continue;
      const startMin=parseInt(m[1])*60+(parseInt(m[2])||0);
      if(startMin>nowMin){nextPeriodData={name:subj,startMin};break;}
    }
  }
  startCountdown();
}

  h+='</colgroup><thead><tr><th class="tt-corner"></th>';
  for(let c=0;c<cols;c++)h+=`<th class="tt-ch" data-col="${c}">${colLetter(c)}<div class="tt-col-resize" data-resize-col="${c}"></div></th>`;
  h+='</tr></thead><tbody>';
  for(let r=0;r<rows;r++){
    h+=`<tr data-row="${r}" style="height:${rowHeights[r]||36}px"><td class="tt-rh" data-rh="${r}">${r+1}<div class="tt-row-resize" data-resize-row="${r}"></div></td>`;
    for(let c=0;c<cols;c++)h+=makeCellHTML(r,c,cells[`${r},${c}`]||{});
    h+='</tr>';
  }
  h+='</tbody></table>';
  el('ttGridWrap').innerHTML=h;
  if(ttSel)highlightTTSel(false);
}

function makeCellHTML(r,c,cell){
  const st=[];
  if(cell.bg)st.push(`background:${cell.bg}`);
  if(cell.fg)st.push(`color:${cell.fg}`);
  if(cell.b)st.push('font-weight:bold');
  if(cell.i)st.push('font-style:italic');
  if(cell.align)st.push(`text-align:${cell.align}`);
  if(cell.fs)st.push(`font-size:${cell.fs}px`);
  const sel=ttSel&&ttSel.r===r&&ttSel.c===c;
  return `<td class="tt-cell${sel?' sel':''}" data-r="${r}" data-c="${c}" style="${st.join(';')}">${escHtml(cell.v||'')}</td>`;
}

function selectTTCell(r,c,scroll=true){
  if(!ttData||r<0||c<0||r>=ttData.rows||c>=ttData.cols)return;
  ttSel={r,c};highlightTTSel(scroll);updateTBState();
}
function highlightTTSel(scroll=true){
  document.querySelectorAll('#ttGrid .sel').forEach(e=>e.classList.remove('sel'));
  document.querySelectorAll('#ttGrid .col-sel').forEach(e=>e.classList.remove('col-sel'));
  document.querySelectorAll('#ttGrid .row-sel').forEach(e=>e.classList.remove('row-sel'));
  if(!ttSel)return;
  const{r,c}=ttSel;
  const ce=getTTCellEl(r,c);if(ce){ce.classList.add('sel');if(scroll)ce.scrollIntoView({block:'nearest',inline:'nearest'});}
  document.querySelector(`#ttGrid .tt-ch[data-col="${c}"]`)?.classList.add('col-sel');
  document.querySelector(`#ttGrid td[data-rh="${r}"]`)?.classList.add('row-sel');
}
function getTTCellEl(r,c){return document.querySelector(`#ttGrid td[data-r="${r}"][data-c="${c}"]`);}
function moveTTSel(dr,dc){if(!ttSel){selectTTCell(0,0);return;}selectTTCell(ttSel.r+dr,ttSel.c+dc);}

function updateTBState(){
  if(!ttSel)return;
  const cell=ttData.cells[`${ttSel.r},${ttSel.c}`]||{};
  el('ttBold')?.classList.toggle('active',!!cell.b);
  el('ttItalic')?.classList.toggle('active',!!cell.i);
  el('ttAlL')?.classList.toggle('active',cell.align==='left');
  el('ttAlC')?.classList.toggle('active',cell.align==='center');
  el('ttAlR')?.classList.toggle('active',cell.align==='right');
  const bg=cell.bg||'#09091f',fg=cell.fg||'#e0e7ff';
  if(el('ttBgPicker')){el('ttBgPicker').value=bg;el('ttBgSwatch').style.background=bg;}
  if(el('ttFgPicker')){el('ttFgPicker').value=fg;el('ttFgSwatch').style.background=fg;}
  if(el('ttFontSize'))el('ttFontSize').value=cell.fs?String(cell.fs):'';
}

function startTTEdit(ch){
  if(!ttSel||ttEditing)return;
  const{r,c}=ttSel;const ce=getTTCellEl(r,c);if(!ce)return;
  ttEditing=true;
  const cell=ttData.cells[`${r},${c}`]||{};
  const inp=document.createElement('input');
  inp.type='text';inp.className='tt-edit-input';
  inp.value=ch!=null?ch:(cell.v||'');
  if(cell.fg)inp.style.color=cell.fg;
  
  ce.textContent='';ce.appendChild(inp);
  inp.focus();
  if(ch!=null)inp.setSelectionRange(inp.value.length,inp.value.length);else inp.select();
  ttEditInput=inp;
}
function commitTTEdit(){
  if(!ttEditing||!ttEditInput||!ttSel){ttEditing=false;ttEditInput=null;return;}
  const{r,c}=ttSel;const key=`${r},${c}`;
  if(!ttData.cells[key])ttData.cells[key]={};
  ttData.cells[key].v=ttEditInput.value;
  ttEditing=false;const inp=ttEditInput;ttEditInput=null;
  if(inp.parentNode){const td=inp.parentNode;inp.remove();applyTTCellStyles(td,ttData.cells[key]);td.textContent=ttData.cells[key].v||'';}
  markTTDirty();
}
function cancelTTEdit(){
  if(!ttEditing||!ttEditInput||!ttSel){ttEditing=false;ttEditInput=null;return;}
  const{r,c}=ttSel;const cell=ttData.cells[`${r},${c}`]||{};
  ttEditing=false;const inp=ttEditInput;ttEditInput=null;
  if(inp.parentNode){const td=inp.parentNode;inp.remove();applyTTCellStyles(td,cell);td.textContent=cell.v||'';}
}
function applyTTCellStyles(td,cell){
  td.style.background=cell.bg||'';td.style.color=cell.fg||'';
  td.style.fontWeight=cell.b?'bold':'';td.style.fontStyle=cell.i?'italic':'';
  td.style.textAlign=cell.align||'';td.style.fontSize=cell.fs?cell.fs+'px':'';
}
function refreshTTCell(r,c){
  const ce=getTTCellEl(r,c);if(!ce)return;
  const cell=ttData.cells[`${r},${c}`]||{};applyTTCellStyles(ce,cell);ce.textContent=cell.v||'';
}
function setTTCellProp(prop,val){
  if(!ttSel)return;
  const key=`${ttSel.r},${ttSel.c}`;
  if(!ttData.cells[key])ttData.cells[key]={};
  if(val===null||val===''||val===undefined)delete ttData.cells[key][prop];else ttData.cells[key][prop]=val;
  refreshTTCell(ttSel.r,ttSel.c);updateTBState();markTTDirty();
}
function toggleTTCellProp(prop){
  if(!ttSel)return;
  const key=`${ttSel.r},${ttSel.c}`;
  if(!ttData.cells[key])ttData.cells[key]={};
  ttData.cells[key][prop]=!ttData.cells[key][prop];
  refreshTTCell(ttSel.r,ttSel.c);updateTBState();markTTDirty();
}
function addTTRow(){commitTTEdit();const at=ttSel?ttSel.r+1:ttData.rows;const nc={};for(const k in ttData.cells){const[r,c]=k.split(',').map(Number);nc[r>=at?`${r+1},${c}`:k]=ttData.cells[k];}ttData.cells=nc;ttData.rows++;ttData.rowHeights.splice(at,0,36);if(ttSel)ttSel={r:at,c:ttSel.c};renderTT();markTTDirty();}
function delTTRow(){if(!ttSel||ttData.rows<=1)return;commitTTEdit();const r=ttSel.r;const nc={};for(const k in ttData.cells){const[rr,c]=k.split(',').map(Number);if(rr===r)continue;nc[rr>r?`${rr-1},${c}`:k]=ttData.cells[k];}ttData.cells=nc;ttData.rows--;ttData.rowHeights.splice(r,1);ttSel={r:Math.min(r,ttData.rows-1),c:ttSel.c};renderTT();markTTDirty();}
function addTTCol(){commitTTEdit();const at=ttSel?ttSel.c+1:ttData.cols;const nc={};for(const k in ttData.cells){const[r,c]=k.split(',').map(Number);nc[c>=at?`${r},${c+1}`:k]=ttData.cells[k];}ttData.cells=nc;ttData.cols++;ttData.colWidths.splice(at,0,120);if(ttSel)ttSel={r:ttSel.r,c:at};renderTT();markTTDirty();}
function delTTCol(){if(!ttSel||ttData.cols<=1)return;commitTTEdit();const c=ttSel.c;const nc={};for(const k in ttData.cells){const[r,cc]=k.split(',').map(Number);if(cc===c)continue;nc[cc>c?`${r},${cc-1}`:k]=ttData.cells[k];}ttData.cells=nc;ttData.cols--;ttData.colWidths.splice(c,1);ttSel={r:ttSel.r,c:Math.min(c,ttData.cols-1)};renderTT();markTTDirty();}

document.addEventListener('keydown',e=>{
  if(!ttPanelOpen)return;
  if(ttEditing){
    if(e.key==='Escape'){cancelTTEdit();e.preventDefault();}
    else if(e.key==='Enter'){commitTTEdit();moveTTSel(1,0);e.preventDefault();}
    else if(e.key==='Tab'){commitTTEdit();moveTTSel(0,e.shiftKey?-1:1);e.preventDefault();}
    return;
  }
  if(e.target.closest('.tt-toolbar'))return;
  if(e.key==='Escape'){el('ttOverlay').classList.add('hide');return;}
  if(!ttSel)return;
  switch(e.key){
    case 'ArrowUp':moveTTSel(-1,0);e.preventDefault();break;
    case 'ArrowDown':moveTTSel(1,0);e.preventDefault();break;
    case 'ArrowLeft':moveTTSel(0,-1);e.preventDefault();break;
    case 'ArrowRight':moveTTSel(0,1);e.preventDefault();break;
    case 'Tab':moveTTSel(0,e.shiftKey?-1:1);e.preventDefault();break;
    case 'Enter':case 'F2':startTTEdit(null);e.preventDefault();break;
    case 'Delete':case 'Backspace':if(ttSel){delete ttData.cells[`${ttSel.r},${ttSel.c}`];refreshTTCell(ttSel.r,ttSel.c);markTTDirty();}break;
    default:if(e.key.length===1&&!e.ctrlKey&&!e.metaKey&&!e.altKey){startTTEdit(e.key);e.preventDefault();}
  }
},true);
